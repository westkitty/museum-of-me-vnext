import { test } from '@playwright/test';
const shots: [string, string, [number, number, number], [number, number, number], [number, number, number]][] = [
  // [template, file, kit position, camera, target]
  ['archive-research-station', 'archive-front', [80, -0.5, -56], [80, 1.7, -49], [80, 0.6, -56]],
  ['archive-research-station', 'archive-oblique', [80, -0.5, -56], [85, 3.2, -51], [80, 0.4, -56]],
  ['archive-research-station', 'archive-chests-close', [80, -0.5, -56], [80.6, 1.5, -53.4], [80, 0.4, -57.3]],
  ['observatory-workstation', 'observatory-front', [80, -0.5, -56], [80, 1.7, -49], [80, 0.6, -56]],
  ['observatory-workstation', 'observatory-oblique', [80, -0.5, -56], [74.5, 3.2, -51], [80, 0.4, -56]],
  ['observatory-workstation', 'observatory-desk-close', [80, -0.5, -56], [80.4, 1.6, -53.6], [80.6, 0.6, -56]],
  ['observatory-workstation', 'observatory-easel-close', [80, -0.5, -56], [76.2, 1.7, -53.4], [78.4, 1.0, -56]],
];
test('render both Exhibit Kit templates in the real runtime (not saved)', async ({ page }) => {
  test.setTimeout(300_000);
  const errors: string[] = []; page.on('pageerror', (e) => errors.push(e.message));
  await page.goto('http://127.0.0.1:5180/?edit=1', { waitUntil: 'commit', timeout: 120_000 });
  await page.waitForFunction(() => Boolean(window.__museum) && Boolean(window.__museumWorkshop), null, { timeout: 120_000 });
  await page.addStyleTag({ content: '#a11y-root, #ui-root, .cinematic-arrival,   .museum-workshop, .dev-flight-sky { display: none !important; }' });
  let placed = '';
  for (const [template, file, pos, cam, target] of shots) {
    if (placed !== template) {
      await page.evaluate(async ({ template, pos }) => {
        const kits = await import(/* @vite-ignore */ '/src/workshop/ExhibitKits.ts');
        const b = window.__museumWorkshop as unknown as { placements: { replaceManifest(m: unknown): unknown } };
        const { manifest } = kits.placeKit({ schemaVersion: 2, objects: [] } as never, template, { position: pos, rotationY: 0, worldId: 'exterior' });
        b.placements.replaceManifest(manifest);
        window.__museum!.player.teleport([pos[0], 0, pos[2] + 7]);
      }, { template, pos });
      placed = template;
      await page.waitForTimeout(6000);
    }
    await page.evaluate(({ cam, target }) => {
      const app = window.__museum!; app.stop();
      app.renderer.camera.position.set(...cam); app.renderer.camera.lookAt(...target);
      app.sky.follow(app.renderer.camera); app.lighting.update(cam); app.exteriorWorld.variableUpdate(0, cam, 'plaza'); app.renderer.render();
    }, { cam, target });
    await page.locator('#museum-canvas').screenshot({ path: `${file}.png` });
    const info = await page.evaluate(() => { const b = window.__museumWorkshop as unknown as { placements: { captureManifest(): { objects: { id: string; label: string; asset?: string; prefab?: string; position: number[] }[] }; residentCount: number } };
      const m = b.placements.captureManifest(); return { resident: b.placements.residentCount, parts: m.objects.map((o) => `${o.label} @ ${o.position.map((v) => v.toFixed(2)).join(',')}`) }; });
    console.log('SHOT', file, JSON.stringify(info));
    await page.evaluate(() => window.__museum!.start());
  }
  await page.evaluate(() => (window.__museumWorkshop as unknown as { placements: { replaceManifest(m: unknown): unknown } }).placements.replaceManifest({ schemaVersion: 2, objects: [] }));
  console.log('ERRORS', errors.length, errors.slice(0, 3).join(' | '));
});
